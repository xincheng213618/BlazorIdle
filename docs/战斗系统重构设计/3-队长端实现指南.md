# 多人战斗系统设计 - 第三部分：队长端实现指南

## 1. 队长端职责概述

### 1.1 核心职责
作为队长，客户端需要：
1. **运行战斗模拟**：使用 `BlazorIdle.Shared` 的战斗引擎
2. **定期同步状态**：每30秒上传战斗快照到服务器
3. **管理队伍**：邀请成员、开始副本、解散队伍
4. **处理离线**：离线前保存状态，上线后恢复控制

### 1.2 不需要做的事
- ❌ 不需要管理队员的客户端连接
- ❌ 不需要实时推送给队员
- ❌ 不需要处理队员的离线/上线

## 2. PartyBattleCoordinator 实现

### 2.1 核心协调器类

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using System.Net.Http;
using System.Net.Http.Json;
using BlazorIdle.Shared.Game;
using BlazorIdle.Shared.DTOs;
using BlazorIdle.Shared.Models;
using Microsoft.Extensions.Logging;

namespace BlazorIdle.Services
{
    /// <summary>
    /// 队长端战斗协调器
    /// 负责运行多人战斗并同步到服务器
    /// </summary>
    public class PartyBattleCoordinator : IDisposable
    {
        private readonly HttpClient _httpClient;
        private readonly IGameConfigService _gameConfig;
        private readonly ILogger<PartyBattleCoordinator> _logger;
        
        private Guid _partyId;
        private string _dungeonId = string.Empty;
        private DungeonManager? _dungeonManager;
        private BattleTeam<Character>? _playerTeam;
        private CancellationTokenSource? _cts;
        
        private const int SyncIntervalMs = 30000;  // 30秒同步一次
        private const int TickMs = 100;            // 战斗tick间隔
        
        // 记录新掉落物（自上次同步以来）
        private readonly List<LootItem> _newLoots = new();
        private readonly object _lootLock = new object();
        
        public PartyBattleCoordinator(
            HttpClient httpClient,
            IGameConfigService gameConfig,
            ILogger<PartyBattleCoordinator> logger)
        {
            _httpClient = httpClient;
            _gameConfig = gameConfig;
            _logger = logger;
        }
        
        /// <summary>
        /// 开始队伍战斗
        /// </summary>
        public async Task<bool> StartPartyBattleAsync(
            Guid partyId,
            string dungeonId,
            List<PartyCharacterData> characters,
            int rngSeed)
        {
            try
            {
                _partyId = partyId;
                _dungeonId = dungeonId;
                
                _logger.LogInformation(
                    "Starting party battle: PartyId={PartyId}, Dungeon={Dungeon}, Members={Count}",
                    partyId, dungeonId, characters.Count);
                
                // 1. 获取副本定义
                var dungeonDef = _gameConfig.GetDungeon(dungeonId);
                if (dungeonDef == null)
                {
                    _logger.LogError("Dungeon not found: {DungeonId}", dungeonId);
                    return false;
                }
                
                // 2. 创建时钟和随机数生成器
                var clock = new SimClock();
                var rng = new RngContext(rngSeed);
                
                // 3. 创建玩家队伍（包含所有队员的角色）
                _playerTeam = new BattleTeam<Character>(
                    "party_team",
                    "队伍",
                    TeamType.Player);
                
                foreach (var charData in characters)
                {
                    var character = CreateCharacterFromData(charData.CharacterData);
                    _playerTeam.AddMember(
                        charData.CharacterId,
                        character,
                        character.MaxHp);
                }
                
                // 4. 创建副本管理器
                _dungeonManager = new DungeonManager(
                    dungeonDef,
                    clock,
                    rng,
                    _playerTeam,
                    _gameConfig);
                
                // 订阅掉落事件
                _dungeonManager.LootDropped += OnLootDropped;
                
                // 5. 启动副本
                _dungeonManager.EnableAutoRepeat(2000);
                _dungeonManager.StartDungeon();
                
                // 6. 启动战斗循环和同步任务
                _cts = new CancellationTokenSource();
                _ = RunBattleLoopAsync(_cts.Token);
                _ = RunSyncLoopAsync(_cts.Token);
                
                return true;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to start party battle");
                return false;
            }
        }
        
        /// <summary>
        /// 战斗循环（本地运行）
        /// </summary>
        private async Task RunBattleLoopAsync(CancellationToken cancellationToken)
        {
            _logger.LogInformation("Battle loop started");
            
            try
            {
                while (!cancellationToken.IsCancellationRequested)
                {
                    await Task.Delay(TickMs, cancellationToken);
                    
                    if (_dungeonManager != null)
                    {
                        _dungeonManager.AdvanceTick(TickMs);
                    }
                }
            }
            catch (TaskCanceledException)
            {
                _logger.LogInformation("Battle loop cancelled");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error in battle loop");
            }
        }
        
        /// <summary>
        /// 同步循环（定期上传快照）
        /// </summary>
        private async Task RunSyncLoopAsync(CancellationToken cancellationToken)
        {
            _logger.LogInformation("Sync loop started");
            
            try
            {
                while (!cancellationToken.IsCancellationRequested)
                {
                    await Task.Delay(SyncIntervalMs, cancellationToken);
                    
                    await SyncBattleSnapshotAsync();
                }
            }
            catch (TaskCanceledException)
            {
                _logger.LogInformation("Sync loop cancelled");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error in sync loop");
            }
        }
        
        /// <summary>
        /// 同步战斗快照到服务器
        /// </summary>
        private async Task SyncBattleSnapshotAsync()
        {
            if (_dungeonManager == null) return;
            
            try
            {
                // 1. 提取当前战斗快照
                var snapshot = ExtractBattleSnapshot();
                
                // 2. 获取新掉落物
                List<LootItem> loots;
                lock (_lootLock)
                {
                    loots = new List<LootItem>(_newLoots);
                    _newLoots.Clear();
                }
                
                // 3. 上传到服务器
                var request = new SyncBattleRequest
                {
                    SnapshotData = snapshot,
                    NewLoots = loots
                };
                
                var response = await _httpClient.PostAsJsonAsync(
                    $"/api/party/{_partyId}/sync-battle",
                    request);
                
                if (response.IsSuccessStatusCode)
                {
                    var result = await response.Content
                        .ReadFromJsonAsync<SyncBattleResponse>();
                    
                    if (result?.IsVerified == true)
                    {
                        _logger.LogDebug("Battle snapshot synced: {SnapshotId}", result.SnapshotId);
                    }
                    else
                    {
                        _logger.LogWarning("Battle snapshot not verified, may switch to hosted mode");
                    }
                }
                else if (response.StatusCode == System.Net.HttpStatusCode.BadRequest)
                {
                    // 检测到异常，服务器接管
                    _logger.LogWarning("Server detected anomaly, switching to hosted mode");
                    await HandleServerTakeoverAsync();
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to sync battle snapshot");
            }
        }
        
        /// <summary>
        /// 提取战斗快照
        /// </summary>
        private PartyBattleSnapshotData ExtractBattleSnapshot()
        {
            if (_dungeonManager == null || _playerTeam == null)
            {
                throw new InvalidOperationException("Battle not initialized");
            }
            
            var dungeonSnapshot = _dungeonManager.GetSnapshot();
            var battleSnapshot = dungeonSnapshot.BattleSnapshot;
            
            var snapshotData = new PartyBattleSnapshotData
            {
                PartyId = _partyId,
                DungeonId = _dungeonId,
                DungeonName = dungeonSnapshot.DungeonName,
                CurrentWaveIndex = dungeonSnapshot.CurrentWaveIndex,
                TotalWaves = dungeonSnapshot.TotalWaves,
                DungeonState = dungeonSnapshot.State.ToString(),
                DungeonElapsedMs = dungeonSnapshot.ElapsedMs,
                CompletionCount = dungeonSnapshot.CompletionCount,
                AutoRepeatEnabled = dungeonSnapshot.AutoRepeatEnabled,
                ClockTimeMs = battleSnapshot?.ElapsedMs ?? 0,
                TotalKills = dungeonSnapshot.TotalKills,
                TotalDeaths = dungeonSnapshot.TotalDeaths,
                TotalLoot = new Dictionary<string, int>(dungeonSnapshot.TotalLoot),
                Timestamp = DateTime.UtcNow,
                GameVersion = "1.0.0"
            };
            
            // 提取角色状态
            foreach (var member in _playerTeam.Members)
            {
                snapshotData.Characters.Add(new PartyCharacterState
                {
                    CharacterId = member.Id,
                    CharacterName = member.Entity.GetType().Name,  // 需要从实际数据获取
                    UserId = 0,  // 需要从映射表获取
                    CurrentHp = member.CurrentHp,
                    MaxHp = member.MaxHp,
                    IsAlive = member.IsAlive,
                    IsOnline = true  // 队长运行时假定所有人在线
                });
            }
            
            // 提取敌人状态（如果有）
            if (dungeonSnapshot.CurrentEnemyTeam != null)
            {
                foreach (var enemy in dungeonSnapshot.CurrentEnemyTeam.Members)
                {
                    snapshotData.Enemies.Add(new PartyEnemyState
                    {
                        EnemyId = enemy.Id,
                        CurrentHp = enemy.CurrentHp,
                        MaxHp = enemy.MaxHp,
                        IsAlive = enemy.IsAlive
                    });
                }
            }
            
            return snapshotData;
        }
        
        /// <summary>
        /// 处理掉落物事件
        /// </summary>
        private void OnLootDropped(LootDropEvent lootEvent)
        {
            lock (_lootLock)
            {
                var existing = _newLoots.FirstOrDefault(l => l.ItemId == lootEvent.ItemId);
                if (existing != null)
                {
                    existing.Quantity += lootEvent.Quantity;
                }
                else
                {
                    _newLoots.Add(new LootItem
                    {
                        ItemId = lootEvent.ItemId,
                        Quantity = lootEvent.Quantity
                    });
                }
            }
        }
        
        /// <summary>
        /// 处理服务器接管
        /// </summary>
        private async Task HandleServerTakeoverAsync()
        {
            _logger.LogInformation("Server taking over battle, stopping local simulation");
            
            // 停止本地战斗
            StopBattle();
            
            // 通知UI
            await Task.CompletedTask;
        }
        
        /// <summary>
        /// 停止战斗
        /// </summary>
        public void StopBattle()
        {
            _logger.LogInformation("Stopping party battle");
            
            _cts?.Cancel();
            
            if (_dungeonManager != null)
            {
                _dungeonManager.LootDropped -= OnLootDropped;
                _dungeonManager.StopDungeon();
            }
        }
        
        /// <summary>
        /// 创建角色实例
        /// </summary>
        private Character CreateCharacterFromData(CharacterData data)
        {
            return new Character
            {
                MaxHp = data.MaxHp,
                Hp = data.MaxHp,
                AttackRateAPS = data.AttackRateAPS,
                DamagePerAttack = data.DamagePerAttack,
                HastePercent = data.HastePercent,
                SpecialIntervalSec = data.SpecialIntervalSec,
                SpecialDamage = data.SpecialDamage,
                CritChancePercent = data.CritChancePercent,
                CritMultiplier = data.CritMultiplier,
                VariancePct = data.VariancePct,
                ReviveMs = (int)(data.ReviveSec * 1000)
            };
        }
        
        public void Dispose()
        {
            StopBattle();
            _cts?.Dispose();
        }
    }
}
```

```razor
    [Parameter]
    public Guid PartyId { get; set; }
    
    private string partyName = "";
    private bool isLeader = false;
    private bool isBattleRunning = false;
    private BattleStatusResponse? battleStatus;
    private List<string> recentLogs = new();
    
    protected override async Task OnInitializedAsync()
    {
        // 加载队伍信息
        await LoadPartyInfoAsync();
        
        // 启动状态轮询
        if (isLeader)
        {
            // 队长：本地运行战斗，轻量轮询
            _ = StartLeaderMonitoringAsync();
        }
        else
        {
            // 队员：60秒轮询服务器
            _ = StartMemberMonitoringAsync();
        }
    }
    
    private async Task StartBattle()
    {
        try
        {
            // 1. 调用API开始副本
            var response = await _httpClient.PostAsJsonAsync(
                $"/api/party/{PartyId}/start-dungeon",
                new StartDungeonRequest { DungeonId = "dungeon_forest_1" });
            
            if (!response.IsSuccessStatusCode)
            {
                Logger.LogError("Failed to start dungeon");
                return;
            }
            
            var startData = await response.Content
                .ReadFromJsonAsync<StartBattleResponse>();
            
            if (startData == null || !startData.Success)
            {
                Logger.LogError("Start battle returned no data");
                return;
            }
            
            // 2. 启动本地战斗协调器
            var success = await BattleCoordinator.StartPartyBattleAsync(
                PartyId,
                startData.DungeonId,
                startData.Characters,
                startData.RngSeed);
            
            if (success)
            {
                isBattleRunning = true;
                Logger.LogInformation("Battle started successfully");
            }
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Error starting battle");
        }
    }
    
    private void StopBattle()
    {
        BattleCoordinator.StopBattle();
        isBattleRunning = false;
    }
    
    private async Task StartLeaderMonitoringAsync()
    {
        // 队长：本地有完整战斗状态，只需要轻量监控
        while (true)
        {
            await Task.Delay(5000);  // 5秒刷新一次UI
            
            if (isBattleRunning)
            {
                // 从本地协调器获取状态
                // battleStatus = BattleCoordinator.GetCurrentStatus();
                StateHasChanged();
            }
        }
    }
    
    private async Task StartMemberMonitoringAsync()
    {
        // 队员：60秒轮询服务器
        while (true)
        {
            await Task.Delay(60000);  // 60秒
            
            try
            {
                var response = await _httpClient.GetAsync(
                    $"/api/party/{PartyId}/battle-status");
                
                if (response.IsSuccessStatusCode)
                {
                    battleStatus = await response.Content
                        .ReadFromJsonAsync<BattleStatusResponse>();
                    
                    if (battleStatus?.RecentLogs != null)
                    {
                        recentLogs = battleStatus.RecentLogs;
                    }
                    
                    StateHasChanged();
                }
            }
            catch (Exception ex)
            {
                Logger.LogError(ex, "Error polling battle status");
            }
        }
    }
    
    private double GetProgressPercent()
    {
        if (battleStatus == null || battleStatus.TotalWaves == 0)
            return 0;
        return (double)battleStatus.CurrentWave / battleStatus.TotalWaves * 100;
    }
    
    private double GetHpPercent(PartyCharacterState character)
    {
        if (character.MaxHp == 0) return 0;
        return (double)character.CurrentHp / character.MaxHp * 100;
    }
    
    private string FormatDuration(int ms)
    {
        var ts = TimeSpan.FromMilliseconds(ms);
        return $"{(int)ts.TotalMinutes:D2}:{ts.Seconds:D2}";
    }
```

## 6. 性能优化建议

### 6.1 本地优化
- 战斗循环使用100ms tick，不需要更高频率
- UI刷新可以降低到500ms或1秒
- 只在战斗状态变化时更新UI

### 6.2 网络优化
- 30秒同步间隔已经足够
- 压缩战斗快照JSON（可选）
- 批量上传掉落物记录

### 6.3 内存优化
- 定期清理战斗日志（保留最近100条）
- 不要在客户端存储完整战斗历史
- 使用对象池复用战斗实例

### 7.1 网络错误
```csharp
private async Task<bool> SyncWithRetryAsync()
{
    const int maxRetries = 3;
    for (int i = 0; i < maxRetries; i++)
    {
        try
        {
            await SyncBattleSnapshotAsync();
            return true;
        }
        catch (Exception ex)
        {
            _logger.LogWarning(ex, "Sync attempt {Attempt} failed", i + 1);
            
            if (i == maxRetries - 1)
            {
                // 最后一次失败，停止本地战斗
                StopBattle();
                return false;
            }
            
            await Task.Delay(1000 * (i + 1));  // 递增延迟
        }
    }
    return false;
}
```

```csharp
private void MonitorBattleHealth()
{
    try
    {
        if (_dungeonManager == null) return;
        
        var snapshot = _dungeonManager.GetSnapshot();
        
        // 检测异常：所有玩家死亡但战斗还在继续
        if (snapshot.BattleSnapshot?.PlayerTeamAlive == 0 
            && snapshot.State == DungeonState.Fighting)
        {
            _logger.LogWarning("Detected anomaly: all players dead but battle running");
            // 通知服务器
            _ = ReportAnomalyAsync();
        }
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error monitoring battle health");
    }
}
```
