#### 1.1.1 GameStateService测试

```csharp
using Xunit;
using Moq;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Configuration;
using BlazorIdle.Server.Services;
using BlazorIdle.Server.Data;
using BlazorIdle.Shared.DTOs;

namespace BlazorIdle.Tests.Services
{
    public class GameStateServiceTests : IDisposable
    {
        private readonly GameDbContext _context;
        private readonly GameStateService _service;
        private readonly Mock<IConfiguration> _mockConfig;
        private readonly Mock<ILogger<GameStateService>> _mockLogger;

        public GameStateServiceTests()
        {
            // 设置内存数据库
            var options = new DbContextOptionsBuilder<GameDbContext>()
                .UseInMemoryDatabase(databaseName: Guid.NewGuid().ToString())
                .Options;

            _context = new GameDbContext(options);
            
            _mockConfig = new Mock<IConfiguration>();
            _mockConfig.Setup(c => c["GameVersion"]).Returns("1.0.0");
            _mockConfig.Setup(c => c.GetValue<int>("GameStateCleanup:RetentionDays", 7)).Returns(7);

            _mockLogger = new Mock<ILogger<GameStateService>>();

            _service = new GameStateService(_context, _mockConfig.Object, _mockLogger.Object);
        }

        [Fact]
        public async Task SaveStateAsync_ShouldCreateNewState_WhenNoExistingState()
        {
            // Arrange
            var characterId = "char_001";
            var stateType = "Battle";
            var stateData = new BattleStateData
            {
                DungeonId = "dungeon_1",
                CurrentWaveIndex = 2
            };

            // Act
            var result = await _service.SaveStateAsync(characterId, stateType, stateData);

            // Assert
            Assert.True(result.Success);
            Assert.NotEqual(Guid.Empty, result.StateId);

            var savedState = await _context.GameStates
                .FirstOrDefaultAsync(gs => gs.CharacterId == characterId && gs.StateType == stateType);

            Assert.NotNull(savedState);
            Assert.Equal("1.0.0", savedState.Version);
        }

        [Fact]
        public async Task SaveStateAsync_ShouldUpdateExistingState_WhenStateExists()
        {
            // Arrange
            var characterId = "char_001";
            var stateType = "Battle";
            
            var initialState = new GameState
            {
                CharacterId = characterId,
                StateType = stateType,
                StateData = "{\"dungeonId\":\"dungeon_1\"}",
                Version = "1.0.0",
                CreatedAt = DateTime.UtcNow.AddHours(-1),
                UpdatedAt = DateTime.UtcNow.AddHours(-1),
                LastHeartbeatAt = DateTime.UtcNow.AddHours(-1)
            };

            _context.GameStates.Add(initialState);
            await _context.SaveChangesAsync();

            var newStateData = new BattleStateData
            {
                DungeonId = "dungeon_2",
                CurrentWaveIndex = 5
            };

            // Act
            var result = await _service.SaveStateAsync(characterId, stateType, newStateData);

            // Assert
            Assert.True(result.Success);
            Assert.Equal(initialState.Id, result.StateId);

            var updatedState = await _context.GameStates.FindAsync(initialState.Id);
            Assert.NotNull(updatedState);
            Assert.Contains("dungeon_2", updatedState.StateData);
            Assert.True(updatedState.UpdatedAt > initialState.UpdatedAt);
        }

        [Fact]
        public async Task LoadStateAsync_ShouldReturnNull_WhenNoStateExists()
        {
            // Act
            var result = await _service.LoadStateAsync("char_999", "Battle");

            // Assert
            Assert.False(result.Success);
            Assert.False(result.HasState);
        }

        [Fact]
        public async Task ValidateStateVersionAsync_ShouldReturnTrue_ForCompatibleVersion()
        {
            // Act
            var result = await _service.ValidateStateVersionAsync("1.0.0");

            // Assert
            Assert.True(result);
        }

        [Fact]
        public async Task ValidateStateVersionAsync_ShouldReturnFalse_ForIncompatibleVersion()
        {
            // Act
            var result = await _service.ValidateStateVersionAsync("2.0.0");

            // Assert
            Assert.False(result);
        }

        [Fact]
        public async Task CleanupExpiredStatesAsync_ShouldRemoveExpiredStates()
        {
            // Arrange
            var expiredState = new GameState
            {
                CharacterId = "char_001",
                StateType = "Battle",
                StateData = "{}",
                Version = "1.0.0",
                ExpiresAt = DateTime.UtcNow.AddDays(-1),
                CreatedAt = DateTime.UtcNow.AddDays(-8),
                UpdatedAt = DateTime.UtcNow.AddDays(-8),
                LastHeartbeatAt = DateTime.UtcNow.AddDays(-8)
            };

            var activeState = new GameState
            {
                CharacterId = "char_002",
                StateType = "Battle",
                StateData = "{}",
                Version = "1.0.0",
                ExpiresAt = DateTime.UtcNow.AddDays(7),
                CreatedAt = DateTime.UtcNow,
                UpdatedAt = DateTime.UtcNow,
                LastHeartbeatAt = DateTime.UtcNow
            };

            _context.GameStates.AddRange(expiredState, activeState);
            await _context.SaveChangesAsync();

            // Act
            await _service.CleanupExpiredStatesAsync();

            // Assert
            var remainingStates = await _context.GameStates.ToListAsync();
            Assert.Single(remainingStates);
            Assert.Equal("char_002", remainingStates[0].CharacterId);
        }

        public void Dispose()
        {
            _context.Dispose();
        }
    }
}
```

#### 1.1.2 SessionService测试

```csharp
using Xunit;
using Moq;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Configuration;
using BlazorIdle.Server.Services;
using BlazorIdle.Server.Data;

namespace BlazorIdle.Tests.Services
{
    public class SessionServiceTests : IDisposable
    {
        private readonly GameDbContext _context;
        private readonly SessionService _service;
        private readonly Mock<IConfiguration> _mockConfig;
        private readonly Mock<ILogger<SessionService>> _mockLogger;

        public SessionServiceTests()
        {
            var options = new DbContextOptionsBuilder<GameDbContext>()
                .UseInMemoryDatabase(databaseName: Guid.NewGuid().ToString())
                .Options;

            _context = new GameDbContext(options);
            
            _mockConfig = new Mock<IConfiguration>();
            _mockConfig.Setup(c => c.GetValue<int>("Session:TimeoutHours", 24)).Returns(24);

            _mockLogger = new Mock<ILogger<SessionService>>();

            _service = new SessionService(_context, _mockConfig.Object, _mockLogger.Object);
        }

        [Fact]
        public async Task CreateSessionAsync_ShouldTerminatePreviousSessions()
        {
            // Arrange
            var userId = 1;
            var existingSession = new UserSession
            {
                UserId = userId,
                SessionToken = "old_token",
                IsActive = true,
                LoginAt = DateTime.UtcNow.AddHours(-1),
                LastActivityAt = DateTime.UtcNow.AddMinutes(-5),
                ExpiresAt = DateTime.UtcNow.AddHours(23)
            };

            _context.UserSessions.Add(existingSession);
            await _context.SaveChangesAsync();

            // Act
            var result = await _service.CreateSessionAsync(userId, "New Device", "192.168.1.100");

            // Assert
            Assert.True(result.Success);
            Assert.True(result.PreviousSessionTerminated);

            var oldSession = await _context.UserSessions.FindAsync(existingSession.Id);
            Assert.NotNull(oldSession);
            Assert.False(oldSession.IsActive);
            Assert.Equal("被新会话顶掉", oldSession.TerminationReason);
        }

        [Fact]
        public async Task HeartbeatAsync_ShouldUpdateLastActivityTime()
        {
            // Arrange
            var session = new UserSession
            {
                UserId = 1,
                SessionToken = "test_token",
                IsActive = true,
                LoginAt = DateTime.UtcNow.AddHours(-1),
                LastActivityAt = DateTime.UtcNow.AddMinutes(-10),
                ExpiresAt = DateTime.UtcNow.AddHours(23)
            };

            _context.UserSessions.Add(session);
            await _context.SaveChangesAsync();

            var originalLastActivity = session.LastActivityAt;
            await Task.Delay(100); // 确保时间有差异

            // Act
            var result = await _service.HeartbeatAsync(session.Id);

            // Assert
            Assert.True(result.Success);
            Assert.True(result.IsActive);

            var updatedSession = await _context.UserSessions.FindAsync(session.Id);
            Assert.NotNull(updatedSession);
            Assert.True(updatedSession.LastActivityAt > originalLastActivity);
        }

        [Fact]
        public async Task HeartbeatAsync_ShouldReturnInactive_ForTerminatedSession()
        {
            // Arrange
            var session = new UserSession
            {
                UserId = 1,
                SessionToken = "test_token",
                IsActive = false,
                TerminatedAt = DateTime.UtcNow.AddMinutes(-5),
                TerminationReason = "被新会话顶掉",
                LoginAt = DateTime.UtcNow.AddHours(-1),
                LastActivityAt = DateTime.UtcNow.AddMinutes(-10),
                ExpiresAt = DateTime.UtcNow.AddHours(23)
            };

            _context.UserSessions.Add(session);
            await _context.SaveChangesAsync();

            // Act
            var result = await _service.HeartbeatAsync(session.Id);

            // Assert
            Assert.False(result.Success);
            Assert.False(result.IsActive);
            Assert.Equal("被新会话顶掉", result.TerminationReason);
        }

        public void Dispose()
        {
            _context.Dispose();
        }
    }
}
```

### 1.2 集成测试

#### 1.2.1 完整流程测试

```csharp
using Xunit;
using Microsoft.AspNetCore.Mvc.Testing;
using System.Net.Http.Json;
using BlazorIdle.Shared.DTOs;

namespace BlazorIdle.Tests.Integration
{
    public class OfflineBattleFlowTests : IClassFixture<WebApplicationFactory<Program>>
    {
        private readonly WebApplicationFactory<Program> _factory;
        private readonly HttpClient _client;

        public OfflineBattleFlowTests(WebApplicationFactory<Program> factory)
        {
            _factory = factory;
            _client = factory.CreateClient();
        }

        [Fact]
        public async Task OfflineBattleFlow_ShouldWorkEndToEnd()
        {
            // 1. 登录
            var loginRequest = new LoginRequest
            {
                Username = "test123",
                Password = "test123123"
            };

            var loginResponse = await _client.PostAsJsonAsync("/api/auth/login", loginRequest);
            Assert.True(loginResponse.IsSuccessStatusCode);

            var authResult = await loginResponse.Content.ReadFromJsonAsync<AuthResponse>();
            Assert.NotNull(authResult);
            Assert.True(authResult.Success);

            // 设置认证令牌
            _client.DefaultRequestHeaders.Authorization = 
                new System.Net.Http.Headers.AuthenticationHeaderValue("Bearer", authResult.Token);

            // 2. 创建角色
            var createCharRequest = new CreateCharacterRequest
            {
                Name = "TestWarrior",
                ProfessionId = "warrior"
            };

            var createCharResponse = await _client.PostAsJsonAsync("/api/character", createCharRequest);
            var charResult = await createCharResponse.Content.ReadFromJsonAsync<CharacterResponse>();
            Assert.NotNull(charResult?.Character);

            var characterId = charResult.Character.Id;

            // 3. 保存游戏状态
            var battleState = new BattleStateData
            {
                DungeonId = "dungeon_forest_1",
                DungeonName = "森林副本",
                CurrentWaveIndex = 2,
                TotalWaves = 5,
                Timestamp = DateTime.UtcNow
            };

            var saveRequest = new GameStateSaveRequest
            {
                CharacterId = characterId,
                StateType = "Battle",
                StateData = battleState
            };

            var saveResponse = await _client.PostAsJsonAsync("/api/gamestate/save", saveRequest);
            Assert.True(saveResponse.IsSuccessStatusCode);

            var saveResult = await saveResponse.Content.ReadFromJsonAsync<GameStateSaveResponse>();
            Assert.NotNull(saveResult);
            Assert.True(saveResult.Success);

            // 4. 模拟离线（等待一段时间）
            await Task.Delay(2000);

            // 5. 加载状态
            var loadResponse = await _client.GetAsync($"/api/gamestate/{characterId}/Battle");
            Assert.True(loadResponse.IsSuccessStatusCode);

            var loadResult = await loadResponse.Content.ReadFromJsonAsync<GameStateLoadResponse>();
            Assert.NotNull(loadResult);
            Assert.True(loadResult.HasState);

            // 6. 模拟离线战斗
            var simulateRequest = new OfflineSimulationRequest
            {
                CharacterId = characterId,
                StateId = loadResult.State!.Id,
                OfflineDurationSeconds = 60
            };

            var simulateResponse = await _client.PostAsJsonAsync(
                "/api/gamestate/simulate-offline",
                simulateRequest);

            Assert.True(simulateResponse.IsSuccessStatusCode);

            var simulateResult = await simulateResponse.Content.ReadFromJsonAsync<OfflineSimulationResult>();
            Assert.NotNull(simulateResult);
            Assert.True(simulateResult.Success);
            Assert.NotNull(simulateResult.Summary);
        }
    }
}
```

```bash
# 配置
DB_PATH="/var/lib/blazoridle/gamedata.db"
BACKUP_DIR="/var/backups/blazoridle"
RETENTION_DAYS=7
```

## 5. 故障排查

### 5.1 常见问题

#### 问题1：数据库锁定

**症状**：出现 "database is locked" 错误

**解决方案**：
1. 检查是否有多个进程访问数据库
2. 增加 SQLite 连接字符串超时时间：
   ```json
   "DefaultConnection": "Data Source=gamedata.db;Timeout=30"
   ```
3. 考虑升级到 PostgreSQL 或 SQL Server

#### 问题2：会话频繁失效

**症状**：用户频繁被踢下线

**解决方案**：
1. 检查会话心跳间隔配置
2. 检查服务器时间是否准确
3. 查看日志确认会话终止原因

#### 问题3：状态恢复失败

**症状**：登录后无法恢复游戏状态

**解决方案**：
1. 检查状态数据版本兼容性
2. 查看服务器日志获取详细错误
3. 验证JSON序列化是否正确

### 5.2 日志分析

重要日志关键词：
- `Session terminated`: 会话被终止
- `Failed to save game state`: 状态保存失败
- `Incompatible state version`: 版本不兼容
- `Cleaned up ... expired`: 清理操作
