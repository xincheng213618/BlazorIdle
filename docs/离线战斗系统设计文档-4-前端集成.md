# 离线战斗系统设计文档 - 第四部分：前端集成与状态管理

## 1. 前端服务架构

**重要说明**：前端服务可以**直接使用** `BlazorIdle.Shared/Game/` 中的战斗引擎进行本地模拟，无需每次都调用服务器。

### 1.1 StateRecoveryService（状态恢复服务）

```csharp
using System;
using System.Net.Http;
using System.Net.Http.Json;
using System.Text.Json;
using System.Threading.Tasks;
using Microsoft.Extensions.Logging;
using BlazorIdle.Shared.Models;
using BlazorIdle.Shared.DTOs;
using BlazorIdle.Shared.Game;              // ← 引用共享的战斗引擎

namespace BlazorIdle.Services
{
    public interface IStateRecoveryService
    {
        Task<StateRecoveryResult> RecoverStateOnLoginAsync(CharacterData character);
        Task<StateRecoveryResult> CatchUpDelayedStateAsync(CharacterData character, TimeSpan timeDelta);
        Task<StateRecoveryResult> SimulateLocallyAsync(BattleStateData savedState, TimeSpan duration);
    }

    /// <summary>
    /// 状态恢复服务
    /// 支持客户端本地模拟（使用 BlazorIdle.Shared 战斗引擎）和服务器端模拟
    /// </summary>
    public class StateRecoveryService : IStateRecoveryService
    {
        private readonly HttpClient _httpClient;
        private readonly ILogger<StateRecoveryService> _logger;
        private readonly IGameConfigService _gameConfig;              // ← 用于本地模拟
        
        private const int MinimumTimeDeltaSeconds = 5;                // 最小时间偏差阈值
        private const int LocalSimulationMaxMinutes = 5;              // 本地模拟最大时长（分钟）

        public StateRecoveryService(
            HttpClient httpClient,
            ILogger<StateRecoveryService> logger)
        {
            _httpClient = httpClient;
            _logger = logger;
        }

        /// <summary>
        /// 登录时恢复游戏状态
        /// </summary>
        public async Task<StateRecoveryResult> RecoverStateOnLoginAsync(CharacterData character)
        {
            try
            {
                _logger.LogInformation("Recovering state for character {CharacterId}", character.Id);

                // 1. 从服务器获取保存的状态
                var response = await _httpClient.GetAsync(
                    $"/api/gamestate/{character.Id}/Battle");

                if (!response.IsSuccessStatusCode)
                {
                    _logger.LogInformation("No saved state found for character {CharacterId}", character.Id);
                    return new StateRecoveryResult
                    {
                        Success = true,
                        HasSavedState = false,
                        Message = "未找到保存的游戏状态，从头开始"
                    };
                }

                var loadResult = await response.Content.ReadFromJsonAsync<GameStateLoadResponse>();
                if (loadResult == null || !loadResult.HasState || loadResult.State == null)
                {
                    return new StateRecoveryResult
                    {
                        Success = true,
                        HasSavedState = false
                    };
                }

                // 2. 计算离线时长
                var savedState = loadResult.State;
                var now = DateTime.UtcNow;
                var lastHeartbeat = savedState.LastHeartbeatAt;
                var offlineDuration = now - lastHeartbeat;

                _logger.LogInformation(
                    "Character {CharacterId} was offline for {Duration}",
                    character.Id,
                    offlineDuration);

                // 如果离线时间很短（< 5秒），直接使用保存的状态，不模拟
                if (offlineDuration.TotalSeconds < MinimumTimeDeltaSeconds)
                {
                    var stateData = JsonSerializer.Deserialize<BattleStateData>(savedState.StateData);
                    return new StateRecoveryResult
                    {
                        Success = true,
                        HasSavedState = true,
                        NeedsSimulation = false,
                        SavedState = stateData,
                        OfflineDuration = offlineDuration
                    };
                }

                // 3. 调用服务器模拟离线战斗
                var simulationRequest = new OfflineSimulationRequest
                {
                    CharacterId = character.Id,
                    StateId = savedState.Id,
                    OfflineDurationSeconds = (int)offlineDuration.TotalSeconds
                };

                var simResponse = await _httpClient.PostAsJsonAsync(
                    "/api/gamestate/simulate-offline",
                    simulationRequest);

                if (!simResponse.IsSuccessStatusCode)
                {
                    _logger.LogWarning("Failed to simulate offline battle for {CharacterId}", character.Id);
                    // 失败时使用保存的状态，不给离线收益
                    var stateData = JsonSerializer.Deserialize<BattleStateData>(savedState.StateData);
                    return new StateRecoveryResult
                    {
                        Success = true,
                        HasSavedState = true,
                        NeedsSimulation = false,
                        SavedState = stateData,
                        Message = "离线模拟失败，使用上次保存的状态"
                    };
                }

                var simulationResult = await simResponse.Content.ReadFromJsonAsync<OfflineSimulationResult>();
                if (simulationResult == null || !simulationResult.Success)
                {
                    var stateData = JsonSerializer.Deserialize<BattleStateData>(savedState.StateData);
                    return new StateRecoveryResult
                    {
                        Success = true,
                        HasSavedState = true,
                        NeedsSimulation = false,
                        SavedState = stateData
                    };
                }

                // 4. 应用模拟结果
                _logger.LogInformation(
                    "Simulation completed: {Kills} kills, {Dungeons} dungeons completed",
                    simulationResult.Summary?.TotalKills ?? 0,
                    simulationResult.Summary?.DungeonsCompleted ?? 0);

                return new StateRecoveryResult
                {
                    Success = true,
                    HasSavedState = true,
                    NeedsSimulation = true,
                    SavedState = simulationResult.NewState,
                    SimulationSummary = simulationResult.Summary,
                    OfflineDuration = offlineDuration,
                    Message = $"离线 {FormatDuration(offlineDuration)}，获得了离线收益"
                };
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to recover state for character {CharacterId}", character.Id);
                return new StateRecoveryResult
                {
                    Success = false,
                    Message = "状态恢复失败：" + ex.Message
                };
            }
        }

        /// <summary>
        /// 追赶时间偏差（切换标签页回来）
        /// </summary>
        public async Task<StateRecoveryResult> CatchUpDelayedStateAsync(
            CharacterData character, 
            TimeSpan timeDelta)
        {
            try
            {
                // 时间偏差小于阈值，不需要追赶
                if (timeDelta.TotalSeconds < MinimumTimeDeltaSeconds)
                {
                    return new StateRecoveryResult
                    {
                        Success = true,
                        NeedsCatchUp = false
                    };
                }

                _logger.LogInformation(
                    "Catching up {Seconds} seconds for character {CharacterId}",
                    timeDelta.TotalSeconds,
                    character.Id);

                // 实现类似离线恢复的逻辑
                // 但这里使用当前的战斗状态作为起点，而不是从数据库加载
                // 具体实现取决于是在客户端还是服务端执行快速追赶

                // 简化版：直接调用服务端模拟
                // 实际项目中可以考虑在客户端本地快速推进以减少网络请求

                return new StateRecoveryResult
                {
                    Success = true,
                    NeedsCatchUp = true,
                    Message = $"已追赶 {FormatDuration(timeDelta)} 的战斗进度"
                };
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to catch up delayed state for character {CharacterId}", character.Id);
                return new StateRecoveryResult
                {
                    Success = false,
                    Message = "追赶失败：" + ex.Message
                };
            }
        }

        private static string FormatDuration(TimeSpan duration)
        {
            if (duration.TotalHours >= 1)
            {
                return $"{(int)duration.TotalHours}小时{duration.Minutes}分钟";
            }
            else if (duration.TotalMinutes >= 1)
            {
                return $"{(int)duration.TotalMinutes}分钟";
            }
            else
            {
                return $"{(int)duration.TotalSeconds}秒";
            }
        }
    }

    public class StateRecoveryResult
    {
        public bool Success { get; set; }
        public bool HasSavedState { get; set; }
        public bool NeedsSimulation { get; set; }
        public bool NeedsCatchUp { get; set; }
        public BattleStateData? SavedState { get; set; }
        public SimulationSummary? SimulationSummary { get; set; }
        public TimeSpan OfflineDuration { get; set; }
        public string? Message { get; set; }
    }
}
```

### 1.2 更新HeartbeatService以保存完整状态

```csharp
using BlazorIdle.Configuration;
using BlazorIdle.Shared.Models;
using BlazorIdle.Shared.DTOs;
using BlazorIdle.Game;
using System.Timers;
using System.Net.Http;
using System.Net.Http.Json;
using Timer = System.Timers.Timer;

namespace BlazorIdle.Services
{
    public class HeartbeatService : IHeartbeatService
    {
        private readonly ICharacterService _characterService;
        private readonly HttpClient _httpClient;
        private readonly HeartbeatConfiguration _config;
        private readonly ILogger<HeartbeatService> _logger;
        
        private Timer? _timer;
        private CharacterData? _currentCharacter;
        private bool _isRunning;
        private readonly object _lock = new object();

        // 新增：用于获取当前战斗状态的委托
        public Func<BattleStateData?>? GetCurrentBattleStateFunc { get; set; }

        public HeartbeatService(
            ICharacterService characterService,
            HttpClient httpClient,
            HeartbeatConfiguration config,
            ILogger<HeartbeatService> logger)
        {
            _characterService = characterService;
            _httpClient = httpClient;
            _config = config;
            _logger = logger;

            _characterService.SelectedCharacterChanged += OnSelectedCharacterChanged;
        }

        public bool IsRunning
        {
            get
            {
                lock (_lock)
                {
                    return _isRunning;
                }
            }
        }

        public void Start(CharacterData character)
        {
            if (character == null)
            {
                _logger.LogWarning("Cannot start heartbeat: character is null");
                return;
            }

            if (!_config.EnableAutoSave)
            {
                _logger.LogInformation("Heartbeat auto-save is disabled in configuration");
                return;
            }

            lock (_lock)
            {
                if (_isRunning)
                {
                    StopInternal();
                }

                _currentCharacter = character;
                
                _timer = new Timer(_config.SaveInterval.TotalMilliseconds);
                _timer.Elapsed += OnTimerElapsed;
                _timer.AutoReset = true;
                _timer.Start();
                
                _isRunning = true;
                
                _logger.LogInformation(
                    $"Heartbeat service started for character '{character.Name}' (ID: {character.Id}), " +
                    $"save interval: {_config.SaveIntervalSeconds} seconds");
            }
        }

        public void Stop()
        {
            lock (_lock)
            {
                StopInternal();
            }
        }

        private void StopInternal()
        {
            if (_timer != null)
            {
                _timer.Stop();
                _timer.Elapsed -= OnTimerElapsed;
                _timer.Dispose();
                _timer = null;
            }

            if (_isRunning)
            {
                _logger.LogInformation(
                    $"Heartbeat service stopped for character '{_currentCharacter?.Name}' " +
                    $"(ID: {_currentCharacter?.Id})");
            }

            _isRunning = false;
            _currentCharacter = null;
        }

        public void UpdateCharacter(CharacterData character)
        {
            lock (_lock)
            {
                if (_isRunning && _currentCharacter != null)
                {
                    _currentCharacter = character;
                }
            }
        }

        public async Task SaveNowAsync()
        {
            CharacterData? characterToSave;
            
            lock (_lock)
            {
                characterToSave = _currentCharacter;
            }

            if (characterToSave != null)
            {
                await SaveCharacterDataAsync(characterToSave);
            }
        }

        private async void OnTimerElapsed(object? sender, ElapsedEventArgs e)
        {
            CharacterData? characterToSave;
            
            lock (_lock)
            {
                characterToSave = _currentCharacter;
            }

            if (characterToSave != null)
            {
                try
                {
                    await SaveCharacterDataAsync(characterToSave);
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Unhandled exception in heartbeat timer elapsed handler");
                }
            }
        }

        private async Task SaveCharacterDataAsync(CharacterData character)
        {
            try
            {
                _logger.LogDebug($"Heartbeat: Saving character '{character.Name}' (ID: {character.Id})");

                // 1. 保存角色数据（现有逻辑）
                var result = await _characterService.UpdateCharacterAsync(character.Id, character);

                if (result?.Success != true)
                {
                    _logger.LogWarning(
                        $"Heartbeat: Failed to save character '{character.Name}': {result?.Message ?? "Unknown error"}");
                }

                // 2. 保存游戏状态（新增）
                var battleState = GetCurrentBattleStateFunc?.Invoke();
                if (battleState != null)
                {
                    await SaveGameStateAsync(character.Id, battleState);
                }
                else
                {
                    _logger.LogDebug("No active battle state to save");
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, $"Heartbeat: Error saving character '{character.Name}' (ID: {character.Id})");
            }
        }

        /// <summary>
        /// 保存游戏状态到服务器
        /// </summary>
        private async Task SaveGameStateAsync(string characterId, BattleStateData stateData)
        {
            try
            {
                var request = new GameStateSaveRequest
                {
                    CharacterId = characterId,
                    StateType = "Battle",
                    StateData = stateData
                };

                var response = await _httpClient.PostAsJsonAsync("/api/gamestate/save", request);

                if (response.IsSuccessStatusCode)
                {
                    var saveResult = await response.Content.ReadFromJsonAsync<GameStateSaveResponse>();
                    _logger.LogDebug(
                        "Saved game state for character {CharacterId}, state ID: {StateId}",
                        characterId,
                        saveResult?.StateId);
                }
                else
                {
                    _logger.LogWarning(
                        "Failed to save game state for character {CharacterId}, status: {Status}",
                        characterId,
                        response.StatusCode);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error saving game state for character {CharacterId}", characterId);
            }
        }

        private void OnSelectedCharacterChanged(CharacterData? character)
        {
            if (character == null)
            {
                Stop();
            }
            else
            {
                Start(character);
            }
        }

        public void Dispose()
        {
            _characterService.SelectedCharacterChanged -= OnSelectedCharacterChanged;
            Stop();
        }
    }
}
```

### 1.3 SessionManagementService（会话管理服务）

```csharp
using System;
using System.Net.Http;
using System.Net.Http.Json;
using System.Threading.Tasks;
using System.Timers;
using Microsoft.Extensions.Logging;
using BlazorIdle.Shared.DTOs;

namespace BlazorIdle.Services
{
    public interface ISessionManagementService
    {
        Task<SessionCreateResponse?> CreateSessionAsync();
        Task StartSessionHeartbeatAsync(Guid sessionId);
        void StopSessionHeartbeat();
        event Action<string>? SessionTerminated;
    }

    public class SessionManagementService : ISessionManagementService
    {
        private readonly HttpClient _httpClient;
        private readonly ILogger<SessionManagementService> _logger;
        private System.Timers.Timer? _heartbeatTimer;
        private Guid _currentSessionId;
        private const int HeartbeatIntervalSeconds = 30;

        public event Action<string>? SessionTerminated;

        public SessionManagementService(
            HttpClient httpClient,
            ILogger<SessionManagementService> logger)
        {
            _httpClient = httpClient;
            _logger = logger;
        }

        public async Task<SessionCreateResponse?> CreateSessionAsync()
        {
            try
            {
                var request = new SessionCreateRequest
                {
                    DeviceInfo = GetDeviceInfo(),
                    IpAddress = null // 由服务器从请求中获取
                };

                var response = await _httpClient.PostAsJsonAsync("/api/session/create", request);

                if (!response.IsSuccessStatusCode)
                {
                    _logger.LogError("Failed to create session, status: {Status}", response.StatusCode);
                    return null;
                }

                var result = await response.Content.ReadFromJsonAsync<SessionCreateResponse>();

                if (result?.Success == true)
                {
                    _logger.LogInformation("Session created: {SessionId}", result.SessionId);
                    
                    if (result.PreviousSessionTerminated)
                    {
                        _logger.LogWarning("Previous session was terminated");
                    }
                }

                return result;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error creating session");
                return null;
            }
        }

        public async Task StartSessionHeartbeatAsync(Guid sessionId)
        {
            StopSessionHeartbeat();

            _currentSessionId = sessionId;
            _heartbeatTimer = new System.Timers.Timer(HeartbeatIntervalSeconds * 1000);
            _heartbeatTimer.Elapsed += async (sender, args) => await SendHeartbeatAsync();
            _heartbeatTimer.AutoReset = true;
            _heartbeatTimer.Start();

            _logger.LogInformation("Session heartbeat started for session {SessionId}", sessionId);

            // 立即发送一次心跳
            await SendHeartbeatAsync();
        }

        public void StopSessionHeartbeat()
        {
            if (_heartbeatTimer != null)
            {
                _heartbeatTimer.Stop();
                _heartbeatTimer.Dispose();
                _heartbeatTimer = null;
                _logger.LogInformation("Session heartbeat stopped");
            }
        }

        private async Task SendHeartbeatAsync()
        {
            try
            {
                var response = await _httpClient.PostAsync(
                    $"/api/session/{_currentSessionId}/heartbeat",
                    null);

                if (response.StatusCode == System.Net.HttpStatusCode.Gone)
                {
                    // 会话已失效（被顶号）
                    var result = await response.Content.ReadFromJsonAsync<SessionHeartbeatResponse>();
                    _logger.LogWarning(
                        "Session terminated: {Reason}",
                        result?.TerminationReason ?? "Unknown");

                    StopSessionHeartbeat();
                    SessionTerminated?.Invoke(result?.TerminationReason ?? "会话已失效");
                }
                else if (!response.IsSuccessStatusCode)
                {
                    _logger.LogWarning(
                        "Session heartbeat failed, status: {Status}",
                        response.StatusCode);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error sending session heartbeat");
            }
        }

        private string GetDeviceInfo()
        {
            // 获取浏览器和操作系统信息
            // 在Blazor WebAssembly中需要使用JSInterop
            return "Blazor WebAssembly"; // 简化版
        }
    }
}
```

```csharp
    /// <summary>
    /// 检测时间偏差并追赶进度（在RunLoopAsync中调用）

    private async Task CheckAndCatchUpTimeDeltaAsync()
    {

        try
        {
            while (!token.IsCancellationRequested && isRunning)
            {
                await Task.Delay(TickMs, token);

                // 检测时间偏差
                await CheckAndCatchUpTimeDeltaAsync();

                // 正常战斗更新
                if (currentBattleMode == BattleMode.Normal)
                {
                    battle!.AdvanceTick(TickMs);
                    snapshot = battle!.GetSnapshot();
                }
                else
                {
                    dungeonManager!.AdvanceTick(TickMs);
                    dungeonSnapshot = dungeonManager.GetSnapshot();

                    if (dungeonSnapshot?.CurrentEnemyTeam != null)
                    {
                        enemyTeam = dungeonSnapshot.CurrentEnemyTeam;

                        if (dungeonManager.CurrentBattle is not null)
                        {
                            battle = dungeonManager.CurrentBattle;
                        }

                        if (dungeonSnapshot.BattleSnapshot != null)
                        {
                            snapshot = dungeonSnapshot.BattleSnapshot;
                        }
                    }
                }

                await InvokeAsync(StateHasChanged);
            }
        }
        catch (TaskCanceledException) { }
        catch (Exception ex)
        {
            _logger?.LogError(ex, "Error in battle loop");
        }
    }

    /// <summary>
    /// 提取当前战斗状态供心跳服务保存
    /// </summary>
    private BattleStateData? ExtractCurrentBattleState()
    {
        if (currentBattleMode != BattleMode.Dungeon || dungeonSnapshot == null)
        {
            return null;
        }

        // 构造战斗状态数据
        var stateData = new BattleStateData
        {
            StateType = "Battle",
            DungeonId = selectedDungeonId,
            DungeonName = currentDungeon?.Name,
            CurrentWaveIndex = dungeonSnapshot.CurrentWaveIndex,
            TotalWaves = dungeonSnapshot.TotalWaves,
            DungeonState = dungeonSnapshot.State.ToString(),
            DungeonElapsedMs = dungeonSnapshot.ElapsedMs,
            CompletionCount = dungeonSnapshot.CompletionCount,
            AutoRepeatEnabled = dungeonSnapshot.AutoRepeatEnabled,
            BattleState = ExtractBattleInstanceState(),
            Timestamp = DateTime.UtcNow,
            GameVersion = "1.0.0"
        };

        return stateData;
    }

    /// <summary>
    /// 提取战斗实例状态
    /// </summary>
    private BattleInstanceState? ExtractBattleInstanceState()
    {
        if (battle == null || snapshot == null) return null;

        // 提取详细的战斗状态
        // 包括：时钟、随机数、队伍状态等
        // 具体实现取决于战斗系统的内部结构

        return new BattleInstanceState
        {
            ClockTimeMs = snapshot.ElapsedMs,
            // ... 其他字段需要从战斗实例中提取
        };
    }

    protected override void OnAfterRender(bool firstRender)
    {
        base.OnAfterRender(firstRender);

        if (firstRender && HeartbeatService != null)
        {
            // 设置心跳服务的状态提取函数
            HeartbeatService.GetCurrentBattleStateFunc = ExtractCurrentBattleState;
        }
    }

    private static string FormatDuration(TimeSpan duration)
    {
        if (duration.TotalHours >= 1)
        {
            return $"{(int)duration.TotalHours}小时{duration.Minutes}分钟";
        }
        else if (duration.TotalMinutes >= 1)
        {
            return $"{(int)duration.TotalMinutes}分钟";
        }
        else
        {
            return $"{(int)duration.TotalSeconds}秒";
        }
    }
}
```

```csharp
@page "/login"
@using BlazorIdle.Services
@inject IAuthService AuthService

        <DataAnnotationsValidator />
        <ValidationSummary />

            <InputText @bind-Value="loginRequest.Username" class="form-control" />
        </div>
        
        <div class="form-group">
            <label>密码：</label>
            <InputText type="password" @bind-Value="loginRequest.Password" class="form-control" />
        </div>
        
        <button type="submit" class="btn btn-primary" disabled="@isLoading">
            @(isLoading ? "登录中..." : "登录")
        </button>
    </EditForm>
</div>

@code {
    private LoginRequest loginRequest = new();
    private string? errorMessage;
    private string? warningMessage;
    private bool isLoading = false;

    private async Task HandleLogin()
    {
        try
        {
            isLoading = true;
            errorMessage = null;
            warningMessage = null;

            // 1. 执行登录
            var result = await AuthService.LoginAsync(loginRequest);

            if (!result.Success)
            {
                errorMessage = result.Message;
                return;
            }

            // 2. 创建会话（实现顶号功能）
            var sessionResult = await SessionService.CreateSessionAsync();

            if (sessionResult == null || !sessionResult.Success)
            {
                errorMessage = "创建会话失败，请重试";
                return;
            }

            // 3. 检查是否顶掉了之前的会话
            if (sessionResult.PreviousSessionTerminated)
            {
                warningMessage = "检测到其他设备的登录已被终止";
            }

            // 4. 启动会话心跳
            await SessionService.StartSessionHeartbeatAsync(sessionResult.SessionId);

            // 5. 订阅会话终止事件
            SessionService.SessionTerminated += OnSessionTerminated;

            // 6. 跳转到主页
            Navigation.NavigateTo("/");
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Login failed");
            errorMessage = "登录失败：" + ex.Message;
        }
        finally
        {
            isLoading = false;
        }
    }

    private void OnSessionTerminated(string reason)
    {
        // 会话被终止（被顶号）
        InvokeAsync(async () =>
        {
            errorMessage = $"您的账号在其他设备登录：{reason}";
            await AuthService.LogoutAsync();
            StateHasChanged();
        });
    }
}
```
