public class CharacterService
{
    private readonly ICharacterAttributeService _attributeService;
    
    // 在构造函数中注入
    public CharacterService(
        // ... 其他依赖
        ICharacterAttributeService attributeService)
    {
        // ... 其他初始化
        _attributeService = attributeService;
    }

    /// <summary>
    /// 加载角色并计算属性
    /// </summary>
    public async Task<CharacterData?> LoadCharacterAsync(string characterId)
    {
        try
        {
            // 1. 从服务器加载角色数据
            var response = await _httpClient.GetAsync($"/api/character/{characterId}");
            
            if (!response.IsSuccessStatusCode)
            {
                _logger.LogWarning("Failed to load character {CharacterId}: {StatusCode}",
                    characterId, response.StatusCode);
                return null;
            }

            var character = await response.Content.ReadFromJsonAsync<CharacterData>();
            
            if (character == null)
            {
                _logger.LogWarning("Character {CharacterId} data is null", characterId);
                return null;
            }

            // 2. 计算并应用属性
            await _attributeService.RecalculateAndApplyAsync(character);

            _logger.LogInformation("Loaded and calculated attributes for character {CharacterId}",
                characterId);

            return character;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error loading character {CharacterId}", characterId);
            return null;
        }
    }

    /// <summary>
    /// 创建新角色并初始化属性
    /// </summary>
    public async Task<CharacterData?> CreateCharacterAsync(string name, string professionId)
    {
        try
        {
            // 1. 创建角色基础数据
            var request = new CreateCharacterRequest
            {
                Name = name,
                ProfessionId = professionId
            };

            var response = await _httpClient.PostAsJsonAsync("/api/character", request);
            
            if (!response.IsSuccessStatusCode)
            {
                _logger.LogWarning("Failed to create character: {StatusCode}",
                    response.StatusCode);
                return null;
            }

            var character = await response.Content.ReadFromJsonAsync<CharacterData>();
            
            if (character == null)
            {
                _logger.LogWarning("Created character data is null");
                return null;
            }

            // 2. 计算并应用初始属性
            await _attributeService.RecalculateAndApplyAsync(character);

            // 3. 保存初始化后的属性
            await UpdateCharacterAsync(character);

            _logger.LogInformation("Created and initialized character {CharacterId}",
                character.Id);

            return character;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error creating character");
            return null;
        }
    }
}
```

public class CharacterService
{
    /// <summary>
    /// 切换激活战斗职业
    /// </summary>
    public async Task<bool> SwitchActiveProfessionAsync(
        CharacterData character, 
        string newProfessionId)
    {
        try
        {
            // 1. 验证新职业是否存在
            if (!character.Professions.ContainsKey(newProfessionId))
            {
                _logger.LogWarning(
                    "Cannot switch to non-existent profession {ProfessionId} for character {CharacterId}",
                    newProfessionId, character.Id);
                return false;
            }

            // 2. 记录旧职业
            var oldProfessionId = character.ActiveCombatProfessionId;

            // 3. 切换职业
            character.ActiveCombatProfessionId = newProfessionId;

            // 4. 重新计算属性
            await _attributeService.RecalculateAndApplyAsync(character);

            _logger.LogInformation(
                "Character {CharacterId} switched profession from {OldProfession} to {NewProfession}",
                character.Id, oldProfessionId, newProfessionId);

            // 5. 保存到服务器（触发一次立即保存）
            await UpdateCharacterAsync(character);

            return true;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex,
                "Error switching profession to {ProfessionId} for character {CharacterId}",
                newProfessionId, character.Id);
            return false;
        }
    }
}

// 在战斗经验获得事件处理中
public class BattleEventHandler
{
    private readonly ICharacterAttributeService _attributeService;
    
    public event EventHandler<CharacterLevelUpEventArgs>? CharacterLeveledUpInCombat;

    public async Task OnExperienceGainedAsync(CharacterData character, int experience)
    {
        // 获取当前职业进度
        var professionId = character.ActiveCombatProfessionId;
        if (!character.Professions.TryGetValue(professionId, out var progress))
        {
            return;
        }

        var oldLevel = progress.Level;
        
        // 添加经验
        progress.Experience += experience;

        // 检查是否升级
        while (progress.Experience >= progress.ExperienceToNext && progress.Level < maxLevel)
        {
            progress.Experience -= progress.ExperienceToNext;
            progress.Level++;
            progress.ExperienceToNext = CalculateNextLevelExp(progress.Level);
        }

        // 如果升级了，触发属性重算和事件
        if (progress.Level > oldLevel)
        {
            _logger.LogInformation(
                "Character {CharacterId} leveled up from {OldLevel} to {NewLevel} during combat",
                character.Id, oldLevel, progress.Level);

            // 重新计算属性
            await _attributeService.RecalculateAndApplyAsync(character);

            // 触发升级事件，通知战斗系统更新角色属性
            CharacterLeveledUpInCombat?.Invoke(this, new CharacterLevelUpEventArgs
            {
                Character = character,
                ProfessionId = professionId,
                OldLevel = oldLevel,
                NewLevel = progress.Level
            });
        }
    }
}

public class CharacterService
{
    /// <summary>
    /// 强制重新计算并应用角色属性
    /// </summary>
    /// <param name="character">要重算的角色</param>
    /// <param name="saveImmediately">是否立即保存到服务器</param>
    public async Task ForceRecalculateAttributesAsync(
        CharacterData character, 
        bool saveImmediately = false)
    {
        try
        {
            _logger.LogInformation(
                "Force recalculating attributes for character {CharacterId}",
                character.Id);

            // 重新计算属性
            await _attributeService.RecalculateAndApplyAsync(character);

            // 如果需要立即保存
            if (saveImmediately)
            {
                await UpdateCharacterAsync(character);
                _logger.LogInformation(
                    "Saved recalculated attributes for character {CharacterId}",
                    character.Id);
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex,
                "Error force recalculating attributes for character {CharacterId}",
                character.Id);
        }
    }

    /// <summary>
    /// 批量重算所有角色的属性（用于配置更新后）
    /// </summary>
    public async Task RecalculateAllCharactersAsync(List<CharacterData> characters)
    {
        _logger.LogInformation(
            "Recalculating attributes for {Count} characters",
            characters.Count);

        foreach (var character in characters)
        {
            await ForceRecalculateAttributesAsync(character, saveImmediately: false);
        }

        _logger.LogInformation("Completed recalculation for all characters");
    }
}
```

3. **状态一致性**：确保计算后立即更新UI
4. **错误容错**：计算失败不应影响主流程
5. **日志完善**：每次触发都要有日志，便于追踪

## 常见问题

**Q: 升级和切换都会触发，是否会重复计算？**
A: 不会。升级只在激活职业时才触发，切换只改变激活职业。这两个操作是互斥的。

**Q: 如果计算失败会怎样？**
A: 角色仍然保留旧的属性值，不会变为null或0。日志会记录错误。

**Q: 是否需要防抖或节流？**
A: MVP阶段不需要。用户操作频率不会太高，且计算很快。

**Q: 切换职业时为什么要立即保存？**
A: 防止用户切换后立即退出导致数据丢失。这是一个重要操作，值得立即保存。
