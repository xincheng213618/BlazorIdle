# 离线战斗系统设计文档 - 第二部分：数据库设计

## 1. 数据库表设计

### 1.1 GameStates 表（游戏状态表）

用于存储各种游戏活动的状态快照。

```sql
CREATE TABLE GameStates (
    Id UNIQUEIDENTIFIER PRIMARY KEY,
    CharacterId NVARCHAR(50) NOT NULL,
    StateType VARCHAR(50) NOT NULL,           -- 状态类型：'Battle', 'Mining', 'Crafting' 等
    StateData NVARCHAR(MAX) NOT NULL,         -- JSON格式的状态数据
    Version NVARCHAR(20) NOT NULL,            -- 软件版本号（如 "1.0.0"）
    SchemaVersion INT NOT NULL DEFAULT 1,     -- 数据结构版本
    CreatedAt DATETIME2 NOT NULL,             -- 创建时间
    UpdatedAt DATETIME2 NOT NULL,             -- 最后更新时间
    LastHeartbeatAt DATETIME2 NOT NULL,       -- 最后心跳时间
    ExpiresAt DATETIME2,                      -- 自动清理时间（可选）
    FOREIGN KEY (CharacterId) REFERENCES Characters(Id) ON DELETE CASCADE
);

-- 索引优化
CREATE INDEX IX_GameStates_CharacterId ON GameStates(CharacterId);
CREATE INDEX IX_GameStates_CharacterId_StateType ON GameStates(CharacterId, StateType);
CREATE INDEX IX_GameStates_ExpiresAt ON GameStates(ExpiresAt) WHERE ExpiresAt IS NOT NULL;
CREATE INDEX IX_GameStates_UpdatedAt ON GameStates(UpdatedAt);
```

#### 字段说明

| 字段 | 类型 | 说明 |
|------|------|------|
| Id | UNIQUEIDENTIFIER | 主键，GUID |
| CharacterId | NVARCHAR(50) | 角色ID，外键引用Characters表 |
| StateType | VARCHAR(50) | 状态类型，用于区分不同的游戏活动 |
| StateData | NVARCHAR(MAX) | JSON格式的状态数据，存储具体的游戏状态 |
| Version | NVARCHAR(20) | 保存状态时的软件版本号 |
| SchemaVersion | INT | 数据结构版本，用于处理数据迁移 |
| CreatedAt | DATETIME2 | 记录创建时间（UTC） |
| UpdatedAt | DATETIME2 | 最后更新时间（UTC） |
| LastHeartbeatAt | DATETIME2 | 最后心跳时间，用于检测离线时长 |
| ExpiresAt | DATETIME2 | 过期时间，用于自动清理 |

### 1.2 UserSessions 表（用户会话表）

用于管理用户登录会话，实现顶号功能。

```sql
CREATE TABLE UserSessions (
    Id UNIQUEIDENTIFIER PRIMARY KEY,
    UserId INT NOT NULL,
    CharacterId NVARCHAR(50),                 -- 当前使用的角色ID（可选）
    SessionToken NVARCHAR(512) NOT NULL,      -- 会话令牌（JWT或自定义）
    DeviceInfo NVARCHAR(500),                 -- 设备信息（可选）
    IpAddress NVARCHAR(50),                   -- IP地址
    LoginAt DATETIME2 NOT NULL,               -- 登录时间
    LastActivityAt DATETIME2 NOT NULL,        -- 最后活动时间
    ExpiresAt DATETIME2 NOT NULL,             -- 会话过期时间
    IsActive BIT NOT NULL DEFAULT 1,          -- 是否活跃
    TerminatedAt DATETIME2,                   -- 被终止时间
    TerminationReason NVARCHAR(200),          -- 终止原因
    FOREIGN KEY (UserId) REFERENCES Users(Id) ON DELETE CASCADE
);

-- 索引优化
CREATE INDEX IX_UserSessions_UserId ON UserSessions(UserId);
CREATE INDEX IX_UserSessions_UserId_IsActive ON UserSessions(UserId, IsActive);
CREATE INDEX IX_UserSessions_SessionToken ON UserSessions(SessionToken);
CREATE INDEX IX_UserSessions_ExpiresAt ON UserSessions(ExpiresAt);
CREATE INDEX IX_UserSessions_LastActivityAt ON UserSessions(LastActivityAt);
```

#### 字段说明

| 字段 | 类型 | 说明 |
|------|------|------|
| Id | UNIQUEIDENTIFIER | 主键，GUID |
| UserId | INT | 用户ID，外键引用Users表 |
| CharacterId | NVARCHAR(50) | 当前使用的角色ID |
| SessionToken | NVARCHAR(512) | 会话令牌，用于识别会话 |
| DeviceInfo | NVARCHAR(500) | 设备信息（如浏览器、OS） |
| IpAddress | NVARCHAR(50) | 登录IP地址 |
| LoginAt | DATETIME2 | 登录时间 |
| LastActivityAt | DATETIME2 | 最后活动时间（心跳更新） |
| ExpiresAt | DATETIME2 | 会话过期时间 |
| IsActive | BIT | 是否活跃（用于顶号标记） |
| TerminatedAt | DATETIME2 | 会话被终止的时间 |
| TerminationReason | NVARCHAR(200) | 终止原因（如"被新会话顶掉"） |

        /// </summary>

        public string Version { get; set; } = "1.0.0";

        /// 数据结构版本

## 4. DbContext 更新

需要在 `GameDbContext.cs` 中添加新表的配置：

```csharp
using Microsoft.EntityFrameworkCore;
using BlazorIdle.Shared.Models;

namespace BlazorIdle.Server.Data
{
    public class GameDbContext : DbContext
    {
        public GameDbContext(DbContextOptions<GameDbContext> options) : base(options)
        {
        }

        public DbSet<User> Users { get; set; }
        public DbSet<CharacterData> Characters { get; set; }
        
        // 新增的表
        public DbSet<GameState> GameStates { get; set; }
        public DbSet<UserSession> UserSessions { get; set; }

        protected override void OnModelCreating(ModelBuilder modelBuilder)
        {
            base.OnModelCreating(modelBuilder);
            
            // 现有配置...（Users, Characters）
            
            // 配置 GameState 实体
            modelBuilder.Entity<GameState>(entity =>
            {
                entity.HasKey(e => e.Id);
                entity.Property(e => e.CharacterId).IsRequired().HasMaxLength(50);
                entity.Property(e => e.StateType).IsRequired().HasMaxLength(50);
                entity.Property(e => e.StateData).IsRequired();
                entity.Property(e => e.Version).IsRequired().HasMaxLength(20);
                entity.Property(e => e.SchemaVersion).HasDefaultValue(1);
                entity.Property(e => e.CreatedAt).IsRequired();
                entity.Property(e => e.UpdatedAt).IsRequired();
                entity.Property(e => e.LastHeartbeatAt).IsRequired();
                
                // 索引
                entity.HasIndex(e => e.CharacterId);
                entity.HasIndex(e => new { e.CharacterId, e.StateType });
                entity.HasIndex(e => e.ExpiresAt).HasFilter("[ExpiresAt] IS NOT NULL");
                entity.HasIndex(e => e.UpdatedAt);
                
                // 外键关系
                entity.HasOne(e => e.Character)
                    .WithMany()
                    .HasForeignKey(e => e.CharacterId)
                    .OnDelete(DeleteBehavior.Cascade);
            });
            
            // 配置 UserSession 实体
            modelBuilder.Entity<UserSession>(entity =>
            {
                entity.HasKey(e => e.Id);
                entity.Property(e => e.UserId).IsRequired();
                entity.Property(e => e.SessionToken).IsRequired().HasMaxLength(512);
                entity.Property(e => e.LoginAt).IsRequired();
                entity.Property(e => e.LastActivityAt).IsRequired();
                entity.Property(e => e.ExpiresAt).IsRequired();
                entity.Property(e => e.IsActive).HasDefaultValue(true);
                
                // 索引
                entity.HasIndex(e => e.UserId);
                entity.HasIndex(e => new { e.UserId, e.IsActive });
                entity.HasIndex(e => e.SessionToken).IsUnique();
                entity.HasIndex(e => e.ExpiresAt);
                entity.HasIndex(e => e.LastActivityAt);
                
                // 外键关系
                entity.HasOne(e => e.User)
                    .WithMany()
                    .HasForeignKey(e => e.UserId)
                    .OnDelete(DeleteBehavior.Cascade);
            });
        }
    }
}
```

## 5. 数据清理策略

### 5.1 过期状态清理

```csharp
/// <summary>
/// 清理过期的游戏状态
/// 建议通过后台任务定期执行（如每小时一次）
/// </summary>
public async Task CleanupExpiredStatesAsync()
{
    var now = DateTime.UtcNow;
    var expiredStates = await _context.GameStates
        .Where(gs => gs.ExpiresAt != null && gs.ExpiresAt < now)
        .ToListAsync();
    
    _context.GameStates.RemoveRange(expiredStates);
    await _context.SaveChangesAsync();
    
    _logger.LogInformation($"Cleaned up {expiredStates.Count} expired game states");
}
```

### 5.2 会话清理

```csharp
/// <summary>
/// 清理过期的会话
/// </summary>
public async Task CleanupExpiredSessionsAsync()
{
    var now = DateTime.UtcNow;
    var expiredSessions = await _context.UserSessions
        .Where(s => s.ExpiresAt < now || 
                    (s.IsActive && s.LastActivityAt < now.AddHours(-24)))
        .ToListAsync();
    
    foreach (var session in expiredSessions)
    {
        session.IsActive = false;
        session.TerminatedAt = now;
        session.TerminationReason = "Session expired";
    }
    
    await _context.SaveChangesAsync();
    
    _logger.LogInformation($"Cleaned up {expiredSessions.Count} expired sessions");
}
```

## 6. 索引优化说明

### 6.1 GameStates 表索引
- `IX_GameStates_CharacterId`：快速查询某个角色的所有状态
- `IX_GameStates_CharacterId_StateType`：快速查询某个角色的特定类型状态（联合索引）
- `IX_GameStates_ExpiresAt`：支持快速清理过期数据（部分索引）
- `IX_GameStates_UpdatedAt`：支持按更新时间排序查询

### 6.2 UserSessions 表索引
- `IX_UserSessions_UserId`：快速查询某用户的所有会话
- `IX_UserSessions_UserId_IsActive`：快速查询某用户的活跃会话（顶号功能）
- `IX_UserSessions_SessionToken`：快速验证会话令牌（唯一索引）
- `IX_UserSessions_ExpiresAt`：支持快速清理过期会话
- `IX_UserSessions_LastActivityAt`：支持查询不活跃会话
